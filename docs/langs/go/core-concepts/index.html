// Generator pattern example
func fibonacci(n int) <-chan int {
    out := make(chan int)
    go func() {
        defer close(out)
        a, b := 0, 1
        for i := 0; i < n; i++ {
            out <- a
            a, b = b, a+b
        }
    }()
    return out
}

// Multiplexing with select
func merge(cs ...<-chan int) <-chan int {
    out := make(chan int)
    var wg sync.WaitGroup
    wg.Add(len(cs))

    for _, c := range cs {
        go func(ch <-chan int) {
            defer wg.Done()
            for n := range ch {
                out <- n
            }
        }(c)
    }

    go func() {
        wg.Wait()
        close(out)
    }()

    return out
}

// Usage example
func main() {
    ch1 := fibonacci(5)
    ch2 := fibonacci(3)
    for n := range merge(ch1, ch2) {
        fmt.Println(n)
    }
}
